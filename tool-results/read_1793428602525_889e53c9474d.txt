    59→    59→  void _startCamera() {
    60→    60→    _cameraController?.dispose();
    61→    61→    _cameraController = mscanner.MobileScannerController(
    62→    62→      detectionSpeed: mscanner.DetectionSpeed.noDuplicates,

    71→    71→    }

    73→    73→      _cameraActive = true;
    74→    74→      _scanResult = null;

    85→    85→  void _onCameraDetect(mscanner.BarcodeCapture capture) {

    88→    88→    if (rawValue == null || rawValue.isEmpty) return;

   111→   111→  Future<void> _pickImageAndScan() async {
   112→   112→    final loc = context.loc;
   113→   113→    setState(() {
   114→   114→      _isScanning = true;
   115→   115→      _scanResult = null;
   116→   116→      _scanError = null;
   117→   117→    });
   118→   118→
   119→   119→    try {
   120→   120→      final result = await FilePicker.platform.pickFiles(
   121→   121→        type: FileType.image,
   122→   122→        allowMultiple: false,
   123→   123→      );
   124→   124→
   125→   125→      if (result == null || result.files.isEmpty) {
   126→   126→        if (mounted) setState(() => _isScanning = false);
   127→   127→        return;
   128→   128→      }
   129→   129→
   130→   130→      final filePath = result.files.first.path;
   131→   131→      if (filePath == null) {
   132→   132→        if (mounted) {
   133→   133→          setState(() {
   134→   134→            _isScanning = false;
   135→   135→            _scanError = loc.t('scanImageError');
   136→   136→          });
   137→   137→        }
   138→   138→        return;
   139→   139→      }
   140→   140→
   141→   141→      final controller = mscanner.MobileScannerController(
   142→   142→        detectionSpeed: mscanner.DetectionSpeed.noDuplicates,
   143→   143→      );
   144→   144→      final capture = await controller.analyzeImage(filePath);
   145→   145→      controller.dispose();
   146→   146→
   147→   147→      if (capture == null || capture.barcodes.isEmpty) {
   148→   148→        if (mounted) {
   149→   149→          setState(() {
   150→   150→            _isScanning = false;
   151→   151→            _scanError = loc.t('noQrFound');
   152→   152→          });
   153→   153→        }
   154→   154→        return;
   155→   155→      }
   156→   156→
   157→   157→      final rawValue = capture.barcodes.first.rawValue;
   158→   158→      if (rawValue == null || rawValue.isEmpty) {
   159→   159→        if (mounted) {
   160→   160→          setState(() {
   161→   161→            _isScanning = false;
   162→   162→            _scanError = loc.t('qrReadError');
   163→   163→          });
   164→   164→        }
   165→   165→        return;
   166→   166→      }
   167→   167→
   168→   168→      _processScanResult(rawValue);
   169→   169→    } catch (e) {
   170→   170→      if (mounted) {
   171→   171→        setState(() {
   172→   172→          _isScanning = false;
   173→   173→          _scanError = '${loc.t('scanFailed')}: $e';
   174→   174→        });
   175→   175→      }
   176→   176→    }
   177→   177→  }

   181→   181→  void _processScanResult(String rawValue) {
   182→   182→    String upiId = '';
   183→   183→    String name = '';
   184→   184→    String amount = '';
   185→   185→    String note = '';
   186→   186→
   187→   187→    if (rawValue.startsWith('upi://')) {
   188→   188→      final uri = Uri.tryParse(rawValue);
   189→   189→      if (uri != null) {
   190→   190→        final params = uri.queryParameters;
   191→   191→        upiId = params['pa'] ?? '';
   192→   192→        name = params['pn'] != null ? Uri.decodeComponent(params['pn']!) : '';
   193→   193→        amount = params['am'] ?? '';
   194→   194→        note = params['tn'] != null ? Uri.decodeComponent(params['tn']!) : '';
   195→   195→      }
   196→   196→    } else if (rawValue.contains('@') && !rawValue.contains(' ')) {
   197→   197→      upiId = rawValue;
   198→   198→    } else {
   199→   199→      // Not a UPI QR — show raw content as upiId placeholder
   200→   200→      upiId = rawValue;
   201→   201→    }
   202→   202→
   203→   203→    final result = {
   204→   204→      'upiId': upiId,
   205→   205→      'name': name,
   206→   206→      'amount': amount,
   207→   207→      'note': note,
   208→   208→      'rawUrl': rawValue.startsWith('upi://') ? rawValue : (upiId.contains('@') ? 'upi://pay?pa=$upiId' : rawValue),
   209→   209→      'timestamp': DateTime.now().toIso8601String(),
   210→   210→    };
   211→   211→
   212→   212→    setState(() {
   213→   213→      _scanResult = result;
   214→   214→      _isScanning = false;
   215→   215→      _scanError = null;
   216→   216→    });
   217→   217→
   218→   218→    // Save to UPI scan history (separate from Counter's QR History)
   219→   219→    try {
   220→   220→      context.read<StorageService>().addUpiScanHistory(result);
   221→   221→    } catch (e) {
   222→   222→      debugPrint('Save UPI scan history failed: $e');
   223→   223→    }
   224→   224→  }

   241→   241→  Future<void> _openUpiApp() async {
   242→   242→    final rawUrl = _scanResult?['rawUrl'];
   243→   243→    if (rawUrl == null) return;
   244→   244→
   245→   245→    String upiLink;
   246→   246→    if (rawUrl.startsWith('upi://')) {
   247→   247→      upiLink = rawUrl;
   248→   248→    } else {
   249→   249→      upiLink = 'upi://pay?pa=$rawUrl';
   250→   250→    }
   251→   251→
   252→   252→    AdMobService.showInterstitial();
   253→   253→
   254→   254→    try {
   255→   255→      final uri = Uri.parse(upiLink);
   256→   256→      final launched = await launchUrl(uri, mode: LaunchMode.externalApplication);
   257→   257→      if (!launched && mounted) {
   258→   258→        final launched2 = await launchUrl(uri);
   259→   259→        if (!launched2 && mounted) {
   260→   260→          await Clipboard.setData(ClipboardData(text: upiLink));
   261→   261→          final loc = context.loc;
   262→   262→          ScaffoldMessenger.of(context).showSnackBar(
   263→   263→            SnackBar(
   264→   264→              content: Text(loc.t('upiLinkCopied')),
   265→   265→              backgroundColor: AppColors.info,
   266→   266→              duration: const Duration(seconds: 4),
   267→   267→            ),
   268→   268→          );
   269→   269→        }
   270→   270→      }
   271→   271→    } catch (e) {
   272→   272→      debugPrint('UPI launch failed: $e');
   273→   273→      await Clipboard.setData(ClipboardData(text: upiLink));
   274→   274→      if (mounted) {
   275→   275→        final loc = context.loc;
   276→   276→        ScaffoldMessenger.of(context).showSnackBar(
   277→   277→          SnackBar(
   278→   278→            content: Text(loc.t('upiLinkCopied')),
   279→   279→            backgroundColor: AppColors.info,
   280→   280→          ),
   281→   281→        );
   282→   282→      }
   283→   283→    }
   284→   284→  }

   286→   286→  void _saveUpiId() {
   287→   287→    if (_scanResult == null) return;
   288→   288→    final upiId = _scanResult!['upiId']!;
   289→   289→    if (upiId.isEmpty) return;
   290→   290→    final name = _scanResult!['name']?.isNotEmpty == true
   291→   291→        ? _scanResult!['name']!
   292→   292→        : (upiId.contains('@') ? upiId.split('@').first : upiId);
   293→   293→    final newUpi = SavedUpi.create(upiId: upiId, name: name, group: 'other');
   294→   294→    SavedUpiService().add(newUpi).then((success) {
   295→   295→      if (!mounted) return;
   296→   296→      final loc = context.loc;
   297→   297→      ScaffoldMessenger.of(context).showSnackBar(
   298→   298→        SnackBar(
   299→   299→          content: Text(success ? loc.t('scanSavedSuccess') : loc.t('upiAlreadyExists')),
   300→   300→          backgroundColor: success ? AppColors.success : AppColors.warning,
   301→   301→        ),
   302→   302→      );
   303→   303→    });
   304→   304→  }
