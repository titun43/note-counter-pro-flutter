     2→// Note Counter Pro - UPI Tools Screen (Premium)
     3→// =============================================================================
     4→// Four-in-one UPI tool (v3.27.0):
     5→//   Tab 1: Generate QR — enter UPI ID/name/amount → generate QR code
     6→//   Tab 2: Scan QR — live camera OR gallery image → decode QR → extract UPI ID
     7→//   Tab 3: Send Money — enter UPI ID + amount → launch UPI deep link
     8→//   Tab 4: Saved UPI — manage saved UPI IDs (Contacts-style)
     9→//
    10→// The Scan feature solves a real problem: UPI QR scan limit is ₹2000
    11→// (same mobile), but UPI ID send limit is ₹1 lakh. By scanning a QR
    12→// via camera or gallery, the user can extract the UPI ID and send
    13→// larger amounts directly from their UPI app.

    96→  void _generateQr() {
    97→    final upiId = _upiIdController.text.trim();
    98→    final loc = context.loc;
    99→    if (upiId.isEmpty) {
   100→      ScaffoldMessenger.of(context).showSnackBar(
   101→        SnackBar(
   102→          content: Text(loc.t('enterUpiId')),
   103→          backgroundColor: AppColors.danger,
   104→        ),
   105→      );
   106→      return;
   107→    }
   108→
   109→    final name = _nameController.text.trim();
   110→    final amount = _amountController.text.trim();
   111→    final note = _noteController.text.trim();
   112→
   113→    final params = <String>[];
   114→    params.add('pa=$upiId');
   115→    if (name.isNotEmpty) params.add('pn=${Uri.encodeComponent(name)}');
   116→    if (amount.isNotEmpty) params.add('am=$amount');
   117→    if (note.isNotEmpty) params.add('tn=${Uri.encodeComponent(note)}');
   118→
   119→    final upiUrl = 'upi://pay?${params.join('&')}';
   120→
   121→    setState(() {
   122→      _generatedUpiUrl = upiUrl;
   123→    });
   124→
   125→    // ✅ Smart save to QR history (Counter → History → QR Code History tab)
   126→    // If UPI ID already exists, update existing entry; else add new entry.
   127→    // Unlimited history.
   128→    try {
   129→      final storage = context.read<StorageService>();
   130→      storage.addQrHistorySmart({
   131→        'upiId': upiId,
   132→        'name': name,
   133→        'amount': amount,
   134→        'note': note,
   135→        'url': upiUrl,
   136→        'timestamp': DateTime.now().toIso8601String(),
   137→      });
   138→      debugPrint('UpiTools: QR saved to history (smart save) for UPI ID: $upiId');
   139→    } catch (e) {
   140→      debugPrint('UpiTools: Failed to save QR to history: $e');
   141→    }
   142→  }

   222→  void _onCameraDetect(mscanner.BarcodeCapture capture) {

   232→  Future<void> _pickImageAndScan() async {
   233→    final loc = context.loc;
   234→    setState(() {
   235→      _isScanning = true;
   236→      _scanResult = null;
   237→      _scanError = null;
   238→    });
   239→
   240→    try {
   241→      final result = await FilePicker.platform.pickFiles(
   242→        type: FileType.image,
   243→        allowMultiple: false,
   244→      );
   245→
   246→      if (result == null || result.files.isEmpty) {
   247→        setState(() => _isScanning = false);
   248→        return;
   249→      }
   250→
   251→      final filePath = result.files.first.path;
   252→      if (filePath == null) {
   253→        setState(() {
   254→          _isScanning = false;
   255→          _scanError = loc.t('scanImageError');
   256→        });
   257→        return;
   258→      }
   259→
   260→      final controller = mscanner.MobileScannerController(
   261→        detectionSpeed: mscanner.DetectionSpeed.noDuplicates,
   262→      );
   263→      final barcode = await controller.analyzeImage(filePath);
   264→      controller.dispose();
   265→
   266→      if (barcode == null || barcode.barcodes.isEmpty) {
   267→        setState(() {
   268→          _isScanning = false;
   269→          _scanError = loc.t('noQrFound');
   270→        });
   271→        return;
   272→      }
   273→
   274→      final rawValue = barcode.barcodes.first.rawValue;
   275→      if (rawValue == null) {
   276→        setState(() {
   277→          _isScanning = false;
   278→          _scanError = loc.t('qrReadError');
   279→        });
   280→        return;
   281→      }
   282→
   283→      _parseUpiString(rawValue);
   284→    } catch (e) {
   285→      setState(() {
   286→        _isScanning = false;
   287→        _scanError = '${loc.t('scanFailed')}: $e';
   288→      });
   289→    }
   290→  }

   292→  /// Parse a UPI deep link string and extract fields.
   293→  void _parseUpiString(String upiString) {
   294→    final result = <String, String>{};
   295→
   296→    if (upiString.startsWith('upi://')) {
   297→      final uri = Uri.tryParse(upiString);
   298→      if (uri != null) {
   299→        final params = uri.queryParameters;
   300→        result['upiId'] = params['pa'] ?? '';
   301→        result['name'] = params['pn'] != null
   302→            ? Uri.decodeComponent(params['pn']!)
   303→            : '';
   304→        result['amount'] = params['am'] ?? '';
   305→        result['note'] = params['tn'] != null
   306→            ? Uri.decodeComponent(params['tn']!)
   307→            : '';
   308→        result['rawUrl'] = upiString;
   309→      }
   310→    } else if (upiString.contains('@') && !upiString.contains(' ')) {
   311→      result['upiId'] = upiString;
   312→      result['name'] = '';
   313→      result['amount'] = '';
   314→      result['note'] = '';
   315→      result['rawUrl'] = upiString;
   316→    } else {
   317→      result['rawUrl'] = upiString;
   318→      result['upiId'] = '';
   319→    }
   320→
   321→    setState(() {
   322→      _scanResult = result;
   323→      _isScanning = false;
   324→      _scanError = null;
   325→      if (result['upiId']!.isNotEmpty) {
   326→        _scanHistory.insert(0, {
   327→          'upiId': result['upiId']!,
   328→          'name': result['name'] ?? '',
   329→          'time': '${DateTime.now().hour}:${DateTime.now().minute.toString().padLeft(2, '0')}',
   330→        });
   331→      }
   332→    });
   333→  }

   348→  Future<void> _openUpiApp() async {
   349→    final rawUrl = _scanResult?['rawUrl'];
   350→    if (rawUrl == null) return;
   351→
   352→    String upiLink;
   353→    if (rawUrl.startsWith('upi://')) {
   354→      upiLink = rawUrl;
   355→    } else {
   356→      upiLink = 'upi://pay?pa=$rawUrl';
   357→    }
   358→
   359→    debugPrint('UpiTools: _openUpiApp — launching UPI link: $upiLink');
   360→
   361→    try {
   362→      final uri = Uri.parse(upiLink);
   363→      // BUG FIX (v3.30.0): Removed canLaunchUrl() check.
   364→      // On Android 11+ package visibility restrictions cause canLaunchUrl()
   365→      // to return false even when UPI apps (PhonePe/GPay/Paytm/BHIM) are
   366→      // installed. This made the button silently fall back to clipboard
   367→      // copy instead of opening any UPI app.
   368→      //
   369→      // Now we directly call launchUrl() with externalApplication mode —
   370→      // the system shows the UPI app chooser if multiple apps are installed,
   371→      // or opens the single installed UPI app directly.
   372→      final launched = await launchUrl(uri, mode: LaunchMode.externalApplication);
   373→
   374→      debugPrint('UpiTools: launchUrl returned = $launched');
   375→
   376→      if (!launched && mounted) {
   377→        // Fallback: try platformDefault mode
   378→        final launched2 = await launchUrl(uri);
   379→        debugPrint('UpiTools: launchUrl fallback returned = $launched2');
   380→        if (!launched2 && mounted) {
   381→          // Last resort: copy to clipboard
   382→          await Clipboard.setData(ClipboardData(text: upiLink));
   383→          final loc = context.loc;
   384→          ScaffoldMessenger.of(context).showSnackBar(
   385→            SnackBar(
   386→              content: Text(loc.t('upiLinkCopied')),
   387→              backgroundColor: AppColors.info,
   388→              duration: const Duration(seconds: 4),
   389→            ),
   390→          );
   391→        }
   392→      }
   393→    } catch (e) {
   394→      debugPrint('UpiTools: _openUpiApp EXCEPTION: $e');
   395→      // Fallback: copy to clipboard
   396→      await Clipboard.setData(ClipboardData(text: upiLink));
   397→      if (mounted) {
   398→        final loc = context.loc;
   399→        ScaffoldMessenger.of(context).showSnackBar(
   400→          SnackBar(
   401→            content: Text(loc.t('upiLinkCopied')),
   402→            backgroundColor: AppColors.info,
   403→          ),
   404→        );
   405→      }
   406→    }
   407→  }
