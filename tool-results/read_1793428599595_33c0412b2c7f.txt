    39→  bool _useCamera = false;

    41→  // Flash state: 0 = OFF, 1 = AUTO, 2 = ON (cycles OFF → AUTO → ON → OFF)
    42→  int _flashMode = 0;
    43→
    44→  mscanner.MobileScannerController? _cameraController;
    45→  bool _cameraActive = false;

    48→  Map<String, String>? _scanResult;

    51→  @override

    57→  // === CAMERA METHODS ===
    58→
    59→  void _startCamera() {
    60→    _cameraController?.dispose();
    61→    _cameraController = mscanner.MobileScannerController(
    62→      detectionSpeed: mscanner.DetectionSpeed.noDuplicates,
    63→      facing: mscanner.CameraFacing.back,
    64→      autoStart: true,
    65→    );
    66→    // Apply torch if flash is ON
    67→    if (_flashMode == 2) {
    68→      Future.delayed(const Duration(milliseconds: 500), () {
    69→        _cameraController?.toggleTorch();
    70→      });
    71→    }
    72→    setState(() {
    73→      _cameraActive = true;
    74→      _scanResult = null;
    75→      _scanError = null;
    76→    });
    77→  }
    78→
    79→  void _stopCamera() {
    80→    _cameraController?.dispose();
    81→    _cameraController = null;
    82→    setState(() => _cameraActive = false);
    83→  }

    85→  void _onCameraDetect(mscanner.BarcodeCapture capture) {
    86→    if (capture.barcodes.isEmpty) return;
    87→    final rawValue = capture.barcodes.first.rawValue;
    88→    if (rawValue == null || rawValue.isEmpty) return;
    89→    _stopCamera();
    90→    _processScanResult(rawValue);
    91→  }
    92→
    93→  /// Cycle flash mode: OFF → AUTO → ON → OFF
    94→  void _toggleFlash() {
    95→    setState(() {
    96→      _flashMode = (_flashMode + 1) % 3;
    97→    });
    98→    if (_cameraActive && _cameraController != null) {
    99→      _cameraController!.toggleTorch();
   100→    }
   101→  }
   102→
   103→  /// Switch camera between front and back
   104→  void _switchCamera() {
   105→    if (_cameraController == null) return;
   106→    _cameraController!.switchCamera();
   107→  }

   110→
   111→  Future<void> _pickImageAndScan() async {
   112→    final loc = context.loc;
   113→    setState(() {
   114→      _isScanning = true;
   115→      _scanResult = null;
   116→      _scanError = null;
   117→    });
   118→
   119→    try {
   120→      final result = await FilePicker.platform.pickFiles(
   121→        type: FileType.image,
   122→        allowMultiple: false,
   123→      );
   124→
   125→      if (result == null || result.files.isEmpty) {
   126→        if (mounted) setState(() => _isScanning = false);
   127→        return;
   128→      }
   129→
   130→      final filePath = result.files.first.path;
   131→      if (filePath == null) {
   132→        if (mounted) {
   133→          setState(() {
   134→            _isScanning = false;
   135→            _scanError = loc.t('scanImageError');
   136→          });
   137→        }
   138→        return;
   139→      }
   140→
   141→      final controller = mscanner.MobileScannerController(
   142→        detectionSpeed: mscanner.DetectionSpeed.noDuplicates,
   143→      );
   144→      final capture = await controller.analyzeImage(filePath);
   145→      controller.dispose();
   146→
   147→      if (capture == null || capture.barcodes.isEmpty) {
   148→        if (mounted) {
   149→          setState(() {
   150→            _isScanning = false;
   151→            _scanError = loc.t('noQrFound');
   152→          });
   153→        }
   154→        return;
   155→      }
   156→
   157→      final rawValue = capture.barcodes.first.rawValue;
   158→      if (rawValue == null || rawValue.isEmpty) {
   159→        if (mounted) {
   160→          setState(() {
   161→            _isScanning = false;
   162→            _scanError = loc.t('qrReadError');
   163→          });
   164→        }
   165→        return;
   166→      }
   167→
   168→      _processScanResult(rawValue);
   169→    } catch (e) {
   170→      if (mounted) {
   171→        setState(() {
   172→          _isScanning = false;
   173→          _scanError = '${loc.t('scanFailed')}: $e';
   174→        });
   175→      }
   176→    }
   177→  }

   179→  // === RESULT PARSER ===
   180→
   181→  void _processScanResult(String rawValue) {
   182→    String upiId = '';
   183→    String name = '';
   184→    String amount = '';
   185→    String note = '';
   186→
   187→    if (rawValue.startsWith('upi://')) {
   188→      final uri = Uri.tryParse(rawValue);
   189→      if (uri != null) {
   190→        final params = uri.queryParameters;
   191→        upiId = params['pa'] ?? '';
   192→        name = params['pn'] != null ? Uri.decodeComponent(params['pn']!) : '';
   193→        amount = params['am'] ?? '';
   194→        note = params['tn'] != null ? Uri.decodeComponent(params['tn']!) : '';
   195→      }
   196→    } else if (rawValue.contains('@') && !rawValue.contains(' ')) {
   197→      upiId = rawValue;
   198→    } else {
   199→      // Not a UPI QR — show raw content as upiId placeholder
   200→      upiId = rawValue;
   201→    }
   202→
   203→    final result = {
   204→      'upiId': upiId,
   205→      'name': name,
   206→      'amount': amount,
   207→      'note': note,
   208→      'rawUrl': rawValue.startsWith('upi://') ? rawValue : (upiId.contains('@') ? 'upi://pay?pa=$upiId' : rawValue),
   209→      'timestamp': DateTime.now().toIso8601String(),
   210→    };
   211→
   212→    setState(() {
   213→      _scanResult = result;
   214→      _isScanning = false;
   215→      _scanError = null;
   216→    });
   217→
   218→    // Save to UPI scan history (separate from Counter's QR History)
   219→    try {
   220→      context.read<StorageService>().addUpiScanHistory(result);
   221→    } catch (e) {
   222→      debugPrint('Save UPI scan history failed: $e');
   223→    }
   224→  }

   228→  void _copyUpiId() {
   229→    final upiId = _scanResult?['upiId'];
   230→    if (upiId == null || upiId.isEmpty) return;
   231→    Clipboard.setData(ClipboardData(text: upiId));
   232→    final loc = context.loc;
   233→    ScaffoldMessenger.of(context).showSnackBar(
   234→      SnackBar(
   235→        content: Text('${loc.t('upiIdCopied')}: $upiId'),
   236→        backgroundColor: AppColors.success,
   237→      ),
   238→    );
   239→  }
   240→
   241→  Future<void> _openUpiApp() async {
   242→    final rawUrl = _scanResult?['rawUrl'];
   243→    if (rawUrl == null) return;
   244→
   245→    String upiLink;
   246→    if (rawUrl.startsWith('upi://')) {
   247→      upiLink = rawUrl;
   248→    } else {
   249→      upiLink = 'upi://pay?pa=$rawUrl';
   250→    }
   251→
   252→    AdMobService.showInterstitial();
   253→
   254→    try {
   255→      final uri = Uri.parse(upiLink);
   256→      final launched = await launchUrl(uri, mode: LaunchMode.externalApplication);
   257→      if (!launched && mounted) {
   258→        final launched2 = await launchUrl(uri);
   259→        if (!launched2 && mounted) {
   260→          await Clipboard.setData(ClipboardData(text: upiLink));
   261→          final loc = context.loc;
   262→          ScaffoldMessenger.of(context).showSnackBar(
   263→            SnackBar(
   264→              content: Text(loc.t('upiLinkCopied')),
   265→              backgroundColor: AppColors.info,
   266→              duration: const Duration(seconds: 4),
   267→            ),
   268→          );
   269→        }
   270→      }
   271→    } catch (e) {
   272→      debugPrint('UPI launch failed: $e');
   273→      await Clipboard.setData(ClipboardData(text: upiLink));
   274→      if (mounted) {
   275→        final loc = context.loc;
   276→        ScaffoldMessenger.of(context).showSnackBar(
   277→          SnackBar(
   278→            content: Text(loc.t('upiLinkCopied')),
   279→            backgroundColor: AppColors.info,
   280→          ),
   281→        );
   282→      }
   283→    }
   284→  }
   285→
   286→  void _saveUpiId() {
   287→    if (_scanResult == null) return;
   288→    final upiId = _scanResult!['upiId']!;
   289→    if (upiId.isEmpty) return;
   290→    final name = _scanResult!['name']?.isNotEmpty == true
   291→        ? _scanResult!['name']!
   292→        : (upiId.contains('@') ? upiId.split('@').first : upiId);
   293→    final newUpi = SavedUpi.create(upiId: upiId, name: name, group: 'other');
   294→    SavedUpiService().add(newUpi).then((success) {
   295→      if (!mounted) return;
   296→      final loc = context.loc;
   297→      ScaffoldMessenger.of(context).showSnackBar(
   298→        SnackBar(
   299→          content: Text(success ? loc.t('scanSavedSuccess') : loc.t('upiAlreadyExists')),
   300→          backgroundColor: success ? AppColors.success : AppColors.warning,
   301→        ),
   302→      );
   303→    });
   304→  }
