     3→     3→// =============================================================================
     4→     4→// Four-in-one UPI tool (v3.27.0):
     5→     5→//   Tab 1: Generate QR — enter UPI ID/name/amount → generate QR code
     6→     6→//   Tab 2: Scan QR — live camera OR gallery image → decode QR → extract UPI ID
     7→     7→//   Tab 3: Send Money — enter UPI ID + amount → launch UPI deep link
     8→     8→//   Tab 4: Saved UPI — manage saved UPI IDs (Contacts-style)
     9→     9→//
    10→    10→// The Scan feature solves a real problem: UPI QR scan limit is ₹2000
    11→    11→// (same mobile), but UPI ID send limit is ₹1 lakh. By scanning a QR
    12→    12→// via camera or gallery, the user can extract the UPI ID and send
    13→    13→// larger amounts directly from their UPI app.

   252→   252→      if (filePath == null) {
   253→   253→        setState(() {
   254→   254→          _isScanning = false;

   262→   262→      );

   264→   264→      controller.dispose();
   265→   265→
   266→   266→      if (barcode == null || barcode.barcodes.isEmpty) {
   267→   267→        setState(() {
   268→   268→          _isScanning = false;
   269→   269→          _scanError = loc.t('noQrFound');
   270→   270→        });
   271→   271→        return;
   272→   272→      }
   273→   273→
   274→   274→      final rawValue = barcode.barcodes.first.rawValue;
   275→   275→      if (rawValue == null) {
   276→   276→        setState(() {
   277→   277→          _isScanning = false;
   278→   278→          _scanError = loc.t('qrReadError');
   279→   279→        });
   280→   280→        return;
   281→   281→      }
   282→   282→
   283→   283→      _parseUpiString(rawValue);
   284→   284→    } catch (e) {
   285→   285→      setState(() {
   286→   286→        _isScanning = false;
   287→   287→        _scanError = '${loc.t('scanFailed')}: $e';
   288→   288→      });
   289→   289→    }

   292→   292→  /// Parse a UPI deep link string and extract fields.
   293→   293→  void _parseUpiString(String upiString) {
   294→   294→    final result = <String, String>{};
   295→   295→
   296→   296→    if (upiString.startsWith('upi://')) {
   297→   297→      final uri = Uri.tryParse(upiString);
   298→   298→      if (uri != null) {
   299→   299→        final params = uri.queryParameters;
   300→   300→        result['upiId'] = params['pa'] ?? '';
   301→   301→        result['name'] = params['pn'] != null
   302→   302→            ? Uri.decodeComponent(params['pn']!)
   303→   303→            : '';
   304→   304→        result['amount'] = params['am'] ?? '';
   305→   305→        result['note'] = params['tn'] != null
   306→   306→            ? Uri.decodeComponent(params['tn']!)
   307→   307→            : '';
   308→   308→        result['rawUrl'] = upiString;
   309→   309→      }
   310→   310→    } else if (upiString.contains('@') && !upiString.contains(' ')) {
   311→   311→      result['upiId'] = upiString;
   312→   312→      result['name'] = '';
   313→   313→      result['amount'] = '';
   314→   314→      result['note'] = '';
   315→   315→      result['rawUrl'] = upiString;
   316→   316→    } else {
   317→   317→      result['rawUrl'] = upiString;
   318→   318→      result['upiId'] = '';
   319→   319→    }
   320→   320→
   321→   321→    setState(() {
   322→   322→      _scanResult = result;
   323→   323→      _isScanning = false;
   324→   324→      _scanError = null;
   325→   325→      if (result['upiId']!.isNotEmpty) {
   326→   326→        _scanHistory.insert(0, {
   327→   327→          'upiId': result['upiId']!,
   328→   328→          'name': result['name'] ?? '',
   329→   329→          'time': '${DateTime.now().hour}:${DateTime.now().minute.toString().padLeft(2, '0')}',
   330→   330→        });
   331→   331→      }
   332→   332→    });
   333→   333→  }

   348→   348→  Future<void> _openUpiApp() async {
   349→   349→    final rawUrl = _scanResult?['rawUrl'];
   350→   350→    if (rawUrl == null) return;
   351→   351→
   352→   352→    String upiLink;
   353→   353→    if (rawUrl.startsWith('upi://')) {
   354→   354→      upiLink = rawUrl;
   355→   355→    } else {
   356→   356→      upiLink = 'upi://pay?pa=$rawUrl';
   357→   357→    }
   358→   358→
   359→   359→    debugPrint('UpiTools: _openUpiApp — launching UPI link: $upiLink');
   360→   360→
   361→   361→    try {
   362→   362→      final uri = Uri.parse(upiLink);
   363→   363→      // BUG FIX (v3.30.0): Removed canLaunchUrl() check.
   364→   364→      // On Android 11+ package visibility restrictions cause canLaunchUrl()
   365→   365→      // to return false even when UPI apps (PhonePe/GPay/Paytm/BHIM) are
   366→   366→      // installed. This made the button silently fall back to clipboard
   367→   367→      // copy instead of opening any UPI app.
   368→   368→      //
   369→   369→      // Now we directly call launchUrl() with externalApplication mode —
   370→   370→      // the system shows the UPI app chooser if multiple apps are installed,
   371→   371→      // or opens the single installed UPI app directly.
   372→   372→      final launched = await launchUrl(uri, mode: LaunchMode.externalApplication);
   373→   373→
   374→   374→      debugPrint('UpiTools: launchUrl returned = $launched');
   375→   375→
   376→   376→      if (!launched && mounted) {
   377→   377→        // Fallback: try platformDefault mode
   378→   378→        final launched2 = await launchUrl(uri);
   379→   379→        debugPrint('UpiTools: launchUrl fallback returned = $launched2');
   380→   380→        if (!launched2 && mounted) {
   381→   381→          // Last resort: copy to clipboard
   382→   382→          await Clipboard.setData(ClipboardData(text: upiLink));
   383→   383→          final loc = context.loc;
   384→   384→          ScaffoldMessenger.of(context).showSnackBar(
   385→   385→            SnackBar(
   386→   386→              content: Text(loc.t('upiLinkCopied')),
   387→   387→              backgroundColor: AppColors.info,
   388→   388→              duration: const Duration(seconds: 4),
   389→   389→            ),
   390→   390→          );
   391→   391→        }
   392→   392→      }
   393→   393→    } catch (e) {
   394→   394→      debugPrint('UpiTools: _openUpiApp EXCEPTION: $e');
   395→   395→      // Fallback: copy to clipboard
   396→   396→      await Clipboard.setData(ClipboardData(text: upiLink));
   397→   397→      if (mounted) {
   398→   398→        final loc = context.loc;
   399→   399→        ScaffoldMessenger.of(context).showSnackBar(
   400→   400→          SnackBar(
   401→   401→            content: Text(loc.t('upiLinkCopied')),
   402→   402→            backgroundColor: AppColors.info,
   403→   403→          ),
   404→   404→        );
   405→   405→      }
   406→   406→    }
   407→   407→  }
